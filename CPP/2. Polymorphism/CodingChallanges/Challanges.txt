Challenge 1 – Create a Cloneable Interface (Virtual Constructor)
Implement:
class Shape {
  public:
    virtual Shape* clone() = 0;
};

Derived:
class Circle : public Shape {
  public:
    int r;
    Circle(int x) : r(x) {}
    Circle* clone() override { return new Circle(*this); }
};
Main should clone shapes into a new vector.

/*----------------------------------------------------------------------------
Challenge Answer: 1 — Cloneable Interface (Virtual Constructor / Prototype)
Goal: Shape::clone() returns a new copy (polymorphic copy).

// cloneable.cpp
#include <iostream>
#include <vector>
#include <memory>
using namespace std;
class Shape {
 public:
    virtual unique_ptr<Shape> clone() const = 0;
    virtual void draw() const = 0;
    virtual ~Shape() = default;
};
class Circle : public Shape {
    int r;
  public:
    Circle(int r_) : r(r_) {}
    unique_ptr<Shape> clone() const override { return make_unique<Circle>(*this); }
    void draw() const override { cout << "Circle r=" << r << "\n"; }
};
class Square : public Shape {
    int side;
  public:
    Square(int s) : side(s) {}
    unique_ptr<Shape> clone() const override { return make_unique<Square>(*this); }
    void draw() const override { cout << "Square side=" << side << "\n"; }
};

int main() {
    vector<unique_ptr<Shape>> shapes;
    shapes.push_back(make_unique<Circle>(5));
    shapes.push_back(make_unique<Square>(3));

    vector<unique_ptr<Shape>> copies;
    for (auto &s : shapes) copies.push_back(s->clone());

    cout << "Originals:\n";
    for (auto &s : shapes) s->draw();

    cout << "Clones:\n";
    for (auto &c : copies) c->draw();
}
/*
Output
Originals:
Circle r=5
Square side=3
Clones:
Circle r=5
Square side=3

//Explanation: clone() is virtual, each derived type returns its own copy. Uses unique_ptr for safe memory management.
-----------------------------------------------------------*/

/*-----------------------------------------------------------
Challenge 2 — Object Slicing and Fix
//Goal: Show slicing and fix it using pointers.

// slicing.cpp
#include <iostream>
#include <vector>
#include <memory>
using namespace std;

struct Base {
    virtual void who() const { cout << "Base\n"; }
    virtual ~Base() = default;
};

struct Derived : public Base {
    int extra;
    Derived(int e=7): extra(e) {}
    void who() const override { cout << "Derived extra=" << extra << "\n"; }
};

int main() {
    cout << "Slicing example:\n";
    vector<Base> v;
    Derived d(42);
    v.push_back(d); // slicing happens
    v[0].who();     // calls Base::who (sliced)

    cout << "\nFixed with pointers:\n";
    vector<unique_ptr<Base>> vp;
    vp.push_back(make_unique<Derived>(42));
    vp[0]->who();   // calls Derived::who
}
/*
Output
Slicing example:
Base

Fixed with pointers:
Derived extra=42

Explanation: Pushing Derived into vector<Base> slices off derived parts. Use vector<unique_ptr<Base>> or references to preserve polymorphism.

Challenge 4 — Visitor Pattern (Double Dispatch)

#include <iostream>
using namespace std;
class Circle;
class Square;

    virtual void visit(Circle*) = 0;
    virtual void visit(Square*) = 0;

    virtual ~Shape() = default;

    void visit(Square* s) override { cout << "Draw Square\n"; s->side(); }
};
int main() {
    Circle c;
    Square s;
    Draw drawer;
    c.accept(&drawer);
    s.accept(&drawer);
}
/*
Output
Draw Circle
Circle radius()
Draw Square
Square side()

Explanation: Double dispatch achieves behavior depending on both visitor and concrete element types.

Write two versions of same code using:

/*-----------------------------------------------------------
Challenge 11 — Template vs Virtual Performance Demo (simple)
Goal: Show two approaches; templated version inlines calls.

// template_vs_virtual.cpp
#include <iostream>
#include <vector>
#include <memory>
using namespace std;
struct Base {
    virtual int op(int) = 0;
    virtual ~Base() = default;
};
struct ImplV : Base {
    int op(int x) override { return x + 1; }
};
template<typename T>
int call_template(T& obj, int n) {
    int sum = 0;
    for (int i=0;i<n;++i) sum += obj.op(i);
    return sum;
}
int call_virtual(Base& b, int n) {
    int sum = 0;
    for (int i=0;i<n;++i) sum += b.op(i);
    return sum;
}
struct ImplT {
    int op(int x) { return x + 1; }
};
int main() {
    ImplT it;
    ImplV iv;

    cout << "template result: " << call_template(it, 5) << "\n";
    cout << "virtual result: " << call_virtual(iv, 5) << "\n";
}
/*
Output
template result: 15
virtual result: 15

Explanation: Both give same result. Templates avoid vptr/indirection and can be inlined; virtual costs a vtable lookup.


-----------------------------------------------------------*/
